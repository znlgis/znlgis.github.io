```csharp
public abstract class BaseEntity
{
    [SugarColumn(IsPrimaryKey = true, IsIdentity = true)]
    public int Id { get; set; }

    [SugarColumn(ColumnDescription = "创建时间")]
    public DateTime CreateTime { get; set; } = DateTime.Now;

    [SugarColumn(ColumnDescription = "创建人", IsNullable = true, Length = 50)]
    public string CreateBy { get; set; }

    [SugarColumn(ColumnDescription = "更新时间", IsNullable = true)]
    public DateTime? UpdateTime { get; set; }

    [SugarColumn(ColumnDescription = "更新人", IsNullable = true, Length = 50)]
    public string UpdateBy { get; set; }

    [SugarColumn(ColumnDescription = "是否删除")]
    public bool IsDeleted { get; set; } = false;
}
```

### 20.2.2 部门实体（树形结构）

```csharp
[SugarTable("Department", TableDescription = "部门表")]
public class Department : BaseEntity
{
    [SugarColumn(Length = 100, ColumnDescription = "部门名称")]
    public string Name { get; set; }

    [SugarColumn(ColumnDescription = "上级部门ID", IsNullable = true)]
    public int? ParentId { get; set; }

    [SugarColumn(ColumnDescription = "排序号")]
    public int Sort { get; set; }

    [SugarColumn(Length = 500, ColumnDescription = "备注", IsNullable = true)]
    public string Remark { get; set; }

    // 导航属性 - 上级部门
    [Navigate(NavigateType.OneToOne, nameof(ParentId))]
    public Department Parent { get; set; }

    // 导航属性 - 下级部门
    [Navigate(NavigateType.OneToMany, nameof(ParentId))]
    public List<Department> Children { get; set; }

    // 导航属性 - 部门员工
    [Navigate(NavigateType.OneToMany, nameof(Employee.DepartmentId))]
    public List<Employee> Employees { get; set; }
}
```

## 20.3 数据库初始化与种子数据

```csharp
public static class DbInitializer
{
    public static void Initialize(ISqlSugarClient db)
    {
        // CodeFirst建表
        db.CodeFirst.InitTables(
            typeof(Department),
            typeof(Employee),
            typeof(Role),
            typeof(EmployeeRole),
            typeof(OperationLog)
        );

        // 初始化种子数据
        if (!db.Queryable<Department>().Any())
        {
            var departments = new List<Department>
            {
                new Department { Id = 1, Name = "总公司", ParentId = null, Sort = 1 },
                new Department { Id = 2, Name = "技术部", ParentId = 1, Sort = 1 },
                new Department { Id = 3, Name = "市场部", ParentId = 1, Sort = 2 },
                new Department { Id = 4, Name = "人事部", ParentId = 1, Sort = 3 },
                new Department { Id = 5, Name = "前端组", ParentId = 2, Sort = 1 },
                new Department { Id = 6, Name = "后端组", ParentId = 2, Sort = 2 },
            };
            db.Insertable(departments).ExecuteCommand();
        }

        if (!db.Queryable<Role>().Any())
        {
            var roles = new List<Role>
            {
                new Role { Name = "管理员", Code = "admin", Description = "系统管理员" },
                new Role { Name = "部门经理", Code = "manager", Description = "部门管理者" },
                new Role { Name = "普通员工", Code = "employee", Description = "普通员工" },
            };
            db.Insertable(roles).ExecuteCommand();
        }
    }
}
```

## 20.4 SqlSugar配置与AOP

```csharp
public static class SqlSugarSetup
{
    public static void AddSqlSugarService(this IServiceCollection services, IConfiguration config)
    {
        services.AddSingleton<ISqlSugarClient>(provider =>
        {
            var logger = provider.GetRequiredService<ILogger<SqlSugarScope>>();

            var db = new SqlSugarScope(new ConnectionConfig()
            {
                ConnectionString = config.GetConnectionString("Default"),
                DbType = DbType.SqlServer,
                IsAutoCloseConnection = true,
                InitKeyType = InitKeyType.Attribute,
                MoreSettings = new ConnMoreSettings()
                {
                    IsAutoRemoveDataCache = true
                }
            }, db =>
            {
                // SQL日志
                db.Aop.OnLogExecuting = (sql, pars) =>
                {
                    logger.LogDebug("[SQL] {Sql}", UtilMethods.GetSqlString(DbType.SqlServer, sql, pars));
                };

                // 慢SQL告警
                db.Aop.OnLogExecuted = (sql, pars) =>
                {
                    if (db.Ado.SqlExecutionTime.TotalSeconds > 2)
                    {
                        logger.LogWarning("[慢SQL] 耗时{Time}ms: {Sql}",
                            db.Ado.SqlExecutionTime.TotalMilliseconds, sql);
                    }
                };

                // 错误日志
                db.Aop.OnError = exp =>
                {
                    logger.LogError(exp, "[SQL异常] {Message}", exp.Message);
                };

                // 自动填充公共字段
                db.Aop.DataExecuting = (oldValue, entityInfo) =>
                {
                    if (entityInfo.OperationType == DataFilterType.InsertByObject)
                    {
                        if (entityInfo.PropertyName == "CreateTime")
                            entityInfo.SetValue(DateTime.Now);
                    }
                    if (entityInfo.OperationType == DataFilterType.UpdateByObject)
                    {
                        if (entityInfo.PropertyName == "UpdateTime")
                            entityInfo.SetValue(DateTime.Now);
                    }
                };

                // 全局软删除过滤
                db.QueryFilter.AddTableFilter<BaseEntity>(it => it.IsDeleted == false);
            });

            // 初始化数据库
            DbInitializer.Initialize(db);

            return db;
        });

        // 注册泛型仓储
        services.AddScoped(typeof(Repository<>));
    }
}
```

## 20.5 员工服务实现

### 20.5.1 员工查询（分页+搜索+导航属性）

```csharp
public class EmployeeService
{
    private readonly ISqlSugarClient _db;

    public EmployeeService(ISqlSugarClient db)
    {
        _db = db;
    }

    /// <summary>
    /// 分页查询员工列表
    /// </summary>
    public PageResult<EmployeeDto> GetPageList(EmployeeQueryDto query)
    {
        int totalCount = 0;

        var list = _db.Queryable<Employee>()
            .Includes(e => e.Department)
            .Includes(e => e.Roles)
            .WhereIF(!string.IsNullOrEmpty(query.Keyword),
                e => e.Name.Contains(query.Keyword) || e.EmployeeNo.Contains(query.Keyword))
            .WhereIF(query.DepartmentId > 0, e => e.DepartmentId == query.DepartmentId)
            .WhereIF(query.Status >= 0, e => e.Status == query.Status)
            .WhereIF(query.HireDateStart.HasValue, e => e.HireDate >= query.HireDateStart.Value)
            .WhereIF(query.HireDateEnd.HasValue, e => e.HireDate <= query.HireDateEnd.Value)
            .OrderBy(e => e.Id, OrderByType.Desc)
            .ToPageList(query.PageIndex, query.PageSize, ref totalCount);

        return new PageResult<EmployeeDto>
        {
            Items = list.Select(e => new EmployeeDto
            {
                Id = e.Id,
                EmployeeNo = e.EmployeeNo,
                Name = e.Name,
                Gender = e.Gender,
                Phone = e.Phone,
                Email = e.Email,
                DepartmentName = e.Department?.Name,
                Position = e.Position,
                HireDate = e.HireDate,
                Salary = e.Salary,
                Status = e.Status,
                Roles = e.Roles?.Select(r => r.Name).ToList()
            }).ToList(),
            TotalCount = totalCount,
            PageIndex = query.PageIndex,
            PageSize = query.PageSize
        };
    }

    /// <summary>
    /// 创建员工（含事务与角色分配）
    /// </summary>
    public int CreateEmployee(EmployeeCreateDto dto)
    {
        // 检查工号唯一
        if (_db.Queryable<Employee>().Any(e => e.EmployeeNo == dto.EmployeeNo))
            throw new Exception($"工号 {dto.EmployeeNo} 已存在");

        var employee = new Employee
        {
            EmployeeNo = dto.EmployeeNo,
            Name = dto.Name,
            Gender = dto.Gender,
            Birthday = dto.Birthday,
            Phone = dto.Phone,
            Email = dto.Email,
            DepartmentId = dto.DepartmentId,
            Position = dto.Position,
            HireDate = dto.HireDate,
            Salary = dto.Salary,
            Status = 1
        };

        int employeeId = 0;
        _db.AsTenant().BeginTran();
        try
        {
            employeeId = _db.Insertable(employee).ExecuteReturnIdentity();

            // 分配角色
            if (dto.RoleIds?.Any() == true)
            {
                var employeeRoles = dto.RoleIds.Select(rid => new EmployeeRole
                {
                    EmployeeId = employeeId,
                    RoleId = rid
                }).ToList();
                _db.Insertable(employeeRoles).ExecuteCommand();
            }

            _db.AsTenant().CommitTran();
        }
        catch
        {
            _db.AsTenant().RollbackTran();
            throw;
        }

        return employeeId;
    }

    /// <summary>
    /// 部门统计
    /// </summary>
    public List<DepartmentStatDto> GetDepartmentStats()
    {
        return _db.Queryable<Employee>()
            .LeftJoin<Department>((e, d) => e.DepartmentId == d.Id)
            .GroupBy((e, d) => new { d.Id, d.Name })
            .Select((e, d) => new DepartmentStatDto
            {
                DepartmentId = d.Id,
                DepartmentName = d.Name,
                EmployeeCount = SqlFunc.AggregateCount(e.Id),
                AvgSalary = SqlFunc.AggregateAvg(e.Salary),
                MaxSalary = SqlFunc.AggregateMax(e.Salary),
                MinSalary = SqlFunc.AggregateMin(e.Salary)
            })
            .ToList();
    }

    /// <summary>
    /// 获取部门树
    /// </summary>
    public List<Department> GetDepartmentTree()
    {
        return _db.Queryable<Department>()
            .ToTree(d => d.Children, d => d.ParentId, null);
    }

    /// <summary>
    /// 批量导入员工
    /// </summary>
    public int BatchImport(List<EmployeeCreateDto> dtos)
    {
        var employees = dtos.Select(dto => new Employee
        {
            EmployeeNo = dto.EmployeeNo,
            Name = dto.Name,
            Gender = dto.Gender,
            DepartmentId = dto.DepartmentId,
            Position = dto.Position,
            HireDate = dto.HireDate,
            Salary = dto.Salary,
            Status = 1
        }).ToList();

        return _db.Fastest<Employee>().BulkCopy(employees);
    }
}
```

## 20.7 本章小结
