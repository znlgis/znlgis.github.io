# 第11章 Go语言扩展开发

PocketBase 不仅是一个独立的后端应用，更是一个功能强大的 Go 框架。开发者可以将 PocketBase 作为 Go 依赖引入自己的项目中，通过编写 Go 代码来深度自定义和扩展 PocketBase 的功能。本章将详细介绍如何使用 Go 语言进行 PocketBase 扩展开发。

---

## 11.1 Go扩展概述

### 11.1.1 为什么选择Go扩展

相比 JavaScript 扩展，Go 扩展具有以下优势：

- **原生性能**：直接运行在 Go 运行时中，无解释器开销
- **完整的 Go 生态**：可以使用任何 Go 第三方库
- **类型安全**：编译时类型检查，减少运行时错误
- **更强的并发能力**：可以利用 goroutine 进行高效并发处理
- **自定义编译**：可以将前端资源嵌入到单一可执行文件中
- **完整 API 访问**：可以访问所有 PocketBase 内部 API 和接口

### 11.1.2 适用场景

Go 扩展特别适合以下场景：

- 需要高性能的数据处理和计算
- 需要集成特定的 Go 第三方库
- 需要自定义认证流程
- 需要将前端与后端打包为单一二进制文件
- 需要添加自定义 CLI 命令
- 需要自定义数据库迁移逻辑

### 11.1.3 架构模式

使用 Go 扩展时，PocketBase 变成了你项目的一个依赖：

```
你的Go应用
├── main.go              # 入口文件，初始化 PocketBase
├── hooks/               # 钩子和业务逻辑
│   ├── records.go
│   └── auth.go
├── routes/              # 自定义路由
│   ├── api.go
│   └── admin.go
├── middleware/           # 中间件
│   └── auth.go
├── services/            # 业务服务
│   └── email.go
├── go.mod
├── go.sum
└── pb_data/             # PocketBase 数据目录
```

```go
func main() {
```

```go
        DefaultDev:     true, // 开发模式

    registerRoutes(app)

func registerHooks(app *pocketbase.PocketBase) {
    app.OnRecordCreate("articles").BindFunc(func(e *core.RecordEvent) error {
        log.Println("新文章创建:", e.Record.GetString("title"))
        return e.Next()
    })
}

func registerRoutes(app *pocketbase.PocketBase) {
    app.OnServe().BindFunc(func(se *core.ServeEvent) error {

            return e.JSON(200, map[string]string{
                "message": "你好，Go扩展！",
            })
        })
        return se.Next()
    })
}
```

## 11.3 自定义API路由

### 11.3.1 注册基本路由

```go
package main

import (
    "log"
    "net/http"

    "github.com/pocketbase/pocketbase"
    "github.com/pocketbase/pocketbase/core"
)

func main() {
    app := pocketbase.New()

    app.OnServe().BindFunc(func(se *core.ServeEvent) error {
        // GET 路由
        se.Router.GET("/api/custom/info", func(e *core.RequestEvent) error {
            return e.JSON(http.StatusOK, map[string]interface{}{
                "name":    "我的应用",
                "version": "1.0.0",
            })
        })

        // POST 路由
        se.Router.POST("/api/custom/submit", func(e *core.RequestEvent) error {
            data := struct {
                Title   string `json:"title"`
                Content string `json:"content"`
            }{}

            if err := e.BindBody(&data); err != nil {
                return e.BadRequestError("请求体解析失败", err)
            }

            if data.Title == "" {
                return e.BadRequestError("标题不能为空", nil)
            }

            // 创建记录
            collection, err := e.App.FindCollectionByNameOrId("articles")
            if err != nil {
                return e.InternalServerError("集合不存在", err)
            }

            record := core.NewRecord(collection)
            record.Set("title", data.Title)
            record.Set("content", data.Content)

            if err := e.App.Save(record); err != nil {
                return e.InternalServerError("保存失败", err)
            }

            return e.JSON(http.StatusCreated, map[string]interface{}{
                "message": "创建成功",
                "id":      record.Id,
            })
        })

        // PUT 路由
        se.Router.PUT("/api/custom/articles/{id}", func(e *core.RequestEvent) error {
            id := e.Request.PathValue("id")

            record, err := e.App.FindRecordById("articles", id)
            if err != nil {
                return e.NotFoundError("文章不存在", err)
            }

            data := struct {
                Title   string `json:"title"`
                Content string `json:"content"`
            }{}

            if err := e.BindBody(&data); err != nil {
                return e.BadRequestError("请求体解析失败", err)
            }

            record.Set("title", data.Title)
            record.Set("content", data.Content)

            if err := e.App.Save(record); err != nil {
                return e.InternalServerError("更新失败", err)
            }

            return e.JSON(http.StatusOK, map[string]string{
                "message": "更新成功",
            })
        })

        // DELETE 路由
        se.Router.DELETE("/api/custom/articles/{id}", func(e *core.RequestEvent) error {
            id := e.Request.PathValue("id")

            record, err := e.App.FindRecordById("articles", id)
            if err != nil {
                return e.NotFoundError("文章不存在", err)
            }

            if err := e.App.Delete(record); err != nil {
                return e.InternalServerError("删除失败", err)
            }

            return e.JSON(http.StatusOK, map[string]string{
                "message": "删除成功",
            })
        })

        return se.Next()
    })

    if err := app.Start(); err != nil {
        log.Fatal(err)
    }
}
```

```go
// 更新之前
app.OnRecordUpdate("articles").BindFunc(func(e *core.RecordEvent) error {
    // 检查是否只有作者可以编辑
    info, _ := e.RequestInfo()
    if info != nil && info.Auth != nil {
        authorId := e.Record.GetString("author")
        if authorId != info.Auth.Id {
            return e.ForbiddenError("只能编辑自己的文章", nil)
        }
    }

    return e.Next()
})

// 更新成功之后
app.OnRecordAfterUpdateSuccess("articles").BindFunc(func(e *core.RecordEvent) error {
    log.Printf("文章已更新: %s", e.Record.Id)

    // 如果文章状态变为已发布，发送通知
    if e.Record.GetString("status") == "published" {
        go sendPublishNotification(e.App, e.Record)
    }

    return e.Next()
})

func sendPublishNotification(app core.App, record *core.Record) {
    authorId := record.GetString("author")
    author, err := app.FindRecordById("users", authorId)
    if err != nil {
        log.Printf("查找作者失败: %v", err)
        return
    }
    log.Printf("通知作者 %s: 文章已发布", author.Email())
}
```

```go
            "-created",
            20,
            0,
            dbx.Params{"userId": "USER_ID"},
        )
        if err != nil {
            return e.InternalServerError("查询失败", err)

        _ = record
```

### 11.5.5 事务操作

```go
func transferBalance(app core.App, senderId, receiverId string, amount float64) error {
    return app.RunInTransaction(func(txApp core.App) error {
        // 查找发送方
        sender, err := txApp.FindRecordById("wallets", senderId)
        if err != nil {
            return fmt.Errorf("发送方不存在: %w", err)
        }

        senderBalance := sender.GetFloat("balance")
        if senderBalance < amount {
            return fmt.Errorf("余额不足")
        }

        // 扣除发送方余额
        sender.Set("balance", senderBalance-amount)
        if err := txApp.Save(sender); err != nil {
            return fmt.Errorf("更新发送方余额失败: %w", err)
        }

        // 查找接收方
        receiver, err := txApp.FindRecordById("wallets", receiverId)
        if err != nil {
            return fmt.Errorf("接收方不存在: %w", err)
        }

        // 增加接收方余额
        receiver.Set("balance", receiver.GetFloat("balance")+amount)
        if err := txApp.Save(receiver); err != nil {
            return fmt.Errorf("更新接收方余额失败: %w", err)
        }

        // 创建转账记录
        collection, err := txApp.FindCollectionByNameOrId("transactions")
        if err != nil {
            return err
        }

        transaction := core.NewRecord(collection)
        transaction.Set("sender", senderId)
        transaction.Set("receiver", receiverId)
        transaction.Set("amount", amount)
        transaction.Set("status", "completed")

        return txApp.Save(transaction)
    })
}
```

### 11.6.1 添加CLI命令

```go
package main

import (
    "fmt"
    "log"

    "github.com/pocketbase/pocketbase"
    "github.com/spf13/cobra"
)

func main() {
    app := pocketbase.New()

    // 添加自定义命令
    app.RootCmd.AddCommand(&cobra.Command{
        Use:   "seed",
        Short: "填充示例数据",
        Run: func(cmd *cobra.Command, args []string) {
            // 必须先初始化应用
            if err := app.Bootstrap(); err != nil {
                log.Fatal(err)
            }

            fmt.Println("开始填充示例数据...")

            collection, err := app.FindCollectionByNameOrId("articles")
            if err != nil {
                log.Fatal("集合不存在:", err)
            }

            articles := []struct {
                Title   string
                Content string
                Status  string
            }{
                {"Go语言入门", "Go语言是一门简洁高效的编程语言...", "published"},
                {"PocketBase教程", "PocketBase是一个开源后端解决方案...", "published"},
                {"数据库设计", "良好的数据库设计是应用成功的基础...", "draft"},
            }

            for _, a := range articles {
                record := core.NewRecord(collection)
                record.Set("title", a.Title)
                record.Set("content", a.Content)
                record.Set("status", a.Status)

                if err := app.Save(record); err != nil {
                    log.Printf("创建失败: %v", err)
                    continue
                }
                fmt.Printf("已创建: %s\n", a.Title)
            }

            fmt.Println("示例数据填充完成！")
        },
    })

    // 添加清理命令
    app.RootCmd.AddCommand(&cobra.Command{
        Use:   "cleanup",
        Short: "清理过期数据",
        Run: func(cmd *cobra.Command, args []string) {
            if err := app.Bootstrap(); err != nil {
                log.Fatal(err)
            }

            fmt.Println("开始清理过期数据...")

            // 清理逻辑
            records, err := app.FindRecordsByFilter(
                "logs",
                "created < {:date}",
                "",
                0, 0,
                dbx.Params{"date": time.Now().AddDate(0, -1, 0).Format(time.RFC3339)},
            )
            if err != nil {
                log.Fatal(err)
            }

            for _, r := range records {
                if err := app.Delete(r); err != nil {
                    log.Printf("删除失败: %v", err)
                }
            }

            fmt.Printf("已清理 %d 条过期记录\n", len(records))
        },
    })

    if err := app.Start(); err != nil {
        log.Fatal(err)
    }
}
```

```go
func APIKeyAuthMiddleware(e *core.RequestEvent) error {
    apiKey := e.Request.Header.Get("X-API-Key")

    if apiKey == "" {
        return e.UnauthorizedError("缺少 API Key", nil)
    }

    // 验证 API Key
    keyRecord, err := e.App.FindFirstRecordByData("api_keys", "key", apiKey)
    if err != nil || !keyRecord.GetBool("active") {
        return e.UnauthorizedError("无效的 API Key", nil)
    }

    // 更新使用统计
    keyRecord.Set("usageCount", keyRecord.GetInt("usageCount")+1)
    keyRecord.Set("lastUsed", time.Now().UTC().Format(time.RFC3339))
    if err := e.App.Save(keyRecord); err != nil {
        log.Printf("更新 API Key 统计失败: %v", err)
    }

    return e.Next()
}
```

```go
        requests: make(map[string][]time.Time),
        max:      max,
        window:   window,
    }
}

func (rl *RateLimiter) Middleware(e *core.RequestEvent) error {
    clientIP := e.RealIP()
    now := time.Now()

    rl.mu.Lock()
    defer rl.mu.Unlock()

    // 清理过期记录
    if times, exists := rl.requests[clientIP]; exists {
        valid := make([]time.Time, 0)
        for _, t := range times {
            if now.Sub(t) < rl.window {
                valid = append(valid, t)
            }
        }
        rl.requests[clientIP] = valid
    }

    // 检查请求次数
    if len(rl.requests[clientIP]) >= rl.max {
        return e.TooManyRequestsError("请求过于频繁，请稍后再试", nil)
    }

    rl.requests[clientIP] = append(rl.requests[clientIP], now)

    return e.Next()
}

// 使用限流中间件
limiter := NewRateLimiter(100, time.Minute)

app.OnServe().BindFunc(func(se *core.ServeEvent) error {
    se.Router.GET("/api/limited/resource", func(e *core.RequestEvent) error {
        return e.JSON(http.StatusOK, map[string]string{"data": "限流保护的资源"})
    }).BindFunc(limiter.Middleware)

    return se.Next()
})
```

```go
func registerAuthRoutes(app *pocketbase.PocketBase) {
    app.OnServe().BindFunc(func(se *core.ServeEvent) error {
        // 手机验证码登录
        se.Router.POST("/api/custom/auth/sms-login", func(e *core.RequestEvent) error {
            data := struct {
                Phone string `json:"phone"`
                Code  string `json:"code"`
            }{}

            if err := e.BindBody(&data); err != nil {
                return e.BadRequestError("请求参数错误", err)
            }

            // 验证验证码
            codeRecord, err := e.App.FindFirstRecordByData("sms_codes", "phone", data.Phone)
            if err != nil {
                return e.BadRequestError("验证码不存在", nil)
            }

            if codeRecord.GetString("code") != data.Code {
                return e.BadRequestError("验证码错误", nil)
            }

            // 检查是否过期（5分钟有效）
            createdAt := codeRecord.GetDateTime("created")
            if time.Since(createdAt.Time()) > 5*time.Minute {
                return e.BadRequestError("验证码已过期", nil)
            }

            // 查找或创建用户
            user, err := e.App.FindFirstRecordByData("users", "phone", data.Phone)
            if err != nil {
                // 用户不存在，自动注册
                collection, _ := e.App.FindCollectionByNameOrId("users")
                user = core.NewRecord(collection)
                user.Set("phone", data.Phone)
                user.Set("username", "user_"+data.Phone[len(data.Phone)-4:])
                if err := e.App.Save(user); err != nil {
                    return e.InternalServerError("用户创建失败", err)
                }
            }

            // 生成认证令牌
            token, err := user.NewAuthToken()
            if err != nil {
                return e.InternalServerError("令牌生成失败", err)
            }

            // 删除已使用的验证码
            _ = e.App.Delete(codeRecord)

            return e.JSON(http.StatusOK, map[string]interface{}{
                "token": token,
                "user":  user,
            })
        })

        // 发送验证码
        se.Router.POST("/api/custom/auth/send-sms", func(e *core.RequestEvent) error {
            data := struct {
                Phone string `json:"phone"`
            }{}

            if err := e.BindBody(&data); err != nil {
                return e.BadRequestError("请求参数错误", err)
            }

            // 生成6位验证码
            code := generateCode(6)

            // 保存验证码
            collection, _ := e.App.FindCollectionByNameOrId("sms_codes")
            record := core.NewRecord(collection)
            record.Set("phone", data.Phone)
            record.Set("code", code)
            if err := e.App.Save(record); err != nil {
                return e.InternalServerError("验证码保存失败", err)
            }

            // TODO: 调用短信 API 发送验证码
            log.Printf("验证码已发送至 %s: %s", data.Phone, code)

            return e.JSON(http.StatusOK, map[string]string{
                "message": "验证码已发送",
            })
        })

        return se.Next()
    })
}

    for i := 0; i < length; i++ {
        n, _ := rand.Int(rand.Reader, big.NewInt(10))
```

### 11.12.3 WebSocket增强

```go
package main

import (
    "log"
    "net/http"
    "sync"

    "github.com/gorilla/websocket"
    "github.com/pocketbase/pocketbase"
    "github.com/pocketbase/pocketbase/core"
)

var upgrader = websocket.Upgrader{
    CheckOrigin: func(r *http.Request) bool {
        return true // 生产环境应做严格检查
    },
}

type Hub struct {
    clients    map[*websocket.Conn]bool
    broadcast  chan []byte
    register   chan *websocket.Conn
    unregister chan *websocket.Conn
    mu         sync.RWMutex
}

func NewHub() *Hub {
    return &Hub{
        clients:    make(map[*websocket.Conn]bool),
        broadcast:  make(chan []byte, 256),
        register:   make(chan *websocket.Conn),
        unregister: make(chan *websocket.Conn),
    }
}

func (h *Hub) Run() {
    for {
        select {
        case conn := <-h.register:
            h.mu.Lock()
            h.clients[conn] = true
            h.mu.Unlock()
            log.Printf("客户端连接，当前连接数: %d", len(h.clients))

        case conn := <-h.unregister:
            h.mu.Lock()
            if _, ok := h.clients[conn]; ok {
                delete(h.clients, conn)
                conn.Close()
            }
            h.mu.Unlock()
            log.Printf("客户端断开，当前连接数: %d", len(h.clients))

        case message := <-h.broadcast:
            h.mu.RLock()
            for conn := range h.clients {
                if err := conn.WriteMessage(websocket.TextMessage, message); err != nil {
                    conn.Close()
                    delete(h.clients, conn)
                }
            }
            h.mu.RUnlock()
        }
    }
}

func (h *Hub) Broadcast(message []byte) {
    h.broadcast <- message
}

func main() {
    app := pocketbase.New()
    hub := NewHub()
    go hub.Run()

    // 注册 WebSocket 路由
    app.OnServe().BindFunc(func(se *core.ServeEvent) error {
        se.Router.GET("/ws", func(e *core.RequestEvent) error {
            conn, err := upgrader.Upgrade(e.Response, e.Request, nil)
            if err != nil {
                return err
            }

            hub.register <- conn

            // 读取消息循环
            go func() {
                defer func() {
                    hub.unregister <- conn
                }()
                for {
                    _, message, err := conn.ReadMessage()
                    if err != nil {
                        break
                    }
                    hub.Broadcast(message)
                }
            }()

            return nil
        })

        return se.Next()
    })

    // 记录变更时广播
    app.OnRecordAfterCreateSuccess().BindFunc(func(e *core.RecordEvent) error {
        message := fmt.Sprintf(`{"event":"create","collection":"%s","id":"%s"}`,
            e.Record.Collection().Name, e.Record.Id)
        hub.Broadcast([]byte(message))
        return e.Next()
    })

    if err := app.Start(); err != nil {
        log.Fatal(err)
    }
}
```

> **小结**：Go 扩展是 PocketBase 最强大的扩展方式，它允许开发者将 PocketBase 作为 Go 框架使用，享受完整的 Go 生态和原生性能。通过自定义路由、事件钩子、中间件、CLI 命令和数据库迁移，可以构建功能丰富的后端应用。结合 `go:embed` 可以将前端资源打包到单一二进制文件中，实现极简部署。对于需要高性能、复杂业务逻辑或深度定制的项目，Go 扩展是最佳选择。
